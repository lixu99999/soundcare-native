## 三、技术方案

### 3.1 核心问题

当前 `hrv-plugin.js`：
```javascript
if (!plugin) {
  return { success: false, errorCode: HRV_ERROR.PLUGIN_NOT_LOADED }
}
```

标准基座下 `uni.requireNativePlugin('SoundCareHRV')` 返回 `null`，所有 HRV 接口**全部失败**，UI 流程跑不通。

### 3.2 解决方案：JS 层 mock 兜底

在 `hrv-plugin.js` 内部检测 `plugin == null` 时，自动切换到 JS 层的 mock 模式：

```javascript
function startMonitoring(options = {}) {
  init()

  if (monitoringActive) {
    return { success: false, errorCode: HRV_ERROR.MONITORING_ALREADY_STARTED }
  }

  if (!plugin) {
    // ✨ 新增：标准基座下启动 JS 层 mock
    startJsMock(options.types)
    monitoringActive = true
    mockModeActive = true
    mockFallbackActive = true  // 标记
    return { success: true, monitoring: true, mockMode: true }
  }

  // ... 原生插件调用（保持不变）
}
```

### 3.3 Mock 数据生成器（沿用小程序 `HRVCalculator` 思路）

> 决策：**不直接复用小程序代码**（Vue 3 setup 语法 vs Options API 不兼容），而是**借鉴同样的算法思路**，在 hrv-plugin.js 内部用更紧凑的实现。

```javascript
// hrv-plugin.js 内部新增（2026-06-15 修正版）
function startJsMock(types) {
  // 借鉴小程序 HRVCalculator：30 秒滚动窗口 + RMSSD
  const rriBuffer = []         // 30 秒窗口内的 RRI 样本
  const windowSize = 30 * 1000
  let baseHR = 70              // 初始 70 BPM（60-80 中位）

  jsMockTimer = setInterval(() => {
    const now = Date.now()

    // 1. 心率 60-80 BPM 随机漂移
    baseHR += (Math.random() - 0.5) * 2
    baseHR = Math.max(60, Math.min(80, baseHR))
    const hr = Math.round(baseHR)

    // 2. 模拟一个 RRI 写入窗口
    //    variation 必须独立于计算出的 RMSSD，否则会形成 feedback loop
    //    让 HRV 一路漂向 0。固定 variation=100 → 期望 RMSSD ≈ 40ms（正常放松）
    const interval = 60000 / hr
    const variation = 100                              // ← 2026-06-15 修复：固定值
    const rri = interval + (Math.random() - 0.5) * variation
    rriBuffer.push({ value: rri, timestamp: now })
    // 清理 30 秒外
    while (rriBuffer.length && rriBuffer[0].timestamp < now - windowSize) {
      rriBuffer.shift()
    }

    // 3. 算 RMSSD（直接基于 RRI 差分，不做 smoothing/feedback）
    let rmssd = MOCK_INITIAL_HRV                       // 默认 40ms
    if (rriBuffer.length >= 2) {
      let sumSq = 0
      for (let i = 0; i < rriBuffer.length - 1; i++) {
        const diff = rriBuffer[i + 1].value - rriBuffer[i].value
        sumSq += diff * diff
      }
      rmssd = Math.sqrt(sumSq / (rriBuffer.length - 1))
    }

    // 4. 推事件
    if (types.includes(HRV_TYPE.HEART_RATE)) {
      handleNativeEvent({
        type: HRV_TYPE.HEART_RATE,
        value: hr,
        timestamp: now,
        source: 'Mock'
      })
    }
    if (types.includes(HRV_TYPE.HRV)) {
      handleNativeEvent({
        type: HRV_TYPE.HRV,
        value: Math.max(0, Math.round(rmssd)),
        timestamp: now,
        source: 'Mock'
      })
    }
  }, 5000)                                              // ← 2026-06-15 修正：5 秒间隔
}
```

**关键点**：
- 算法思路和小程序 HRVCalculator 一致（30 秒窗口、RMSSD、心率漂移）
- 实现更紧凑（30 行 vs 小程序 120 行）
- 事件通过同一个 `handleNativeEvent` 路径推送
- `source: 'Mock'` 在 UI 上可见
- `music.js` 的 `startHRVSession` 收到 source='Mock' 时，**上报后端用 `device_type: 'simulated'`**

**2026-06-15 修正历史**：
- ❌ **原 bug**：`variation = targetHRV / 2` + `targetHRV += (rmssd - targetHRV) * 0.1` 反馈回路
  - 现象：HRV 从初始 40ms 一路下降到 10ms 以下（焦虑状态），无法稳定
  - 原因：RMSSD 算出后会被 `targetHRV` 拉低，targetHRV 又作为下一轮 variation 基数，恶性循环
- ✅ **修复**：`variation = 100`（固定值），移除 targetHRV 反馈
  - 效果：RMSSD 自然围绕 40ms 漂移（30-50ms 区间，落在「中度压力」到「正常放松」之间）
  - 副作用：30 秒窗口满之前（启动后 0-30s），HRV 会从默认 40ms 缓慢漂到真实计算值（轻微跳变）

### 11.1 核心组件

微信小程序用了一个完整的 **`HRVCalculator` 类**（约 120 行），比 v1.1 草案的简单随机游走要精细得多：

```javascript
// soundcare-app/src/pages/player/index.vue:34
class HRVCalculator {
  constructor() {
    this.windowSize = 30 * 1000  // 30秒滚动窗口
    this.rriBuffer = []         // 原始 RRI（毫秒）
    this.heartRateBuffer = []   // 心率历史
  }

  // 从 RRI 计算 RMSSD（标准心率变异算法）
  calculateRMSSD() {
    if (this.rriBuffer.length < 2) return null
    const rris = this.rriBuffer.map(d => d.value)
    let sumSq = 0
    for (let i = 0; i < rris.length - 1; i++) {
      const diff = rris[i + 1] - rris[i]
      sumSq += diff * diff
    }
    return Math.sqrt(sumSq / (rris.length - 1))
  }

  // 从心率变异系数（CV = std/avg）估算状态
  estimateStatusFromHeartRate() {
    if (this.heartRateBuffer.length < 10) return null
    const rates = this.heartRateBuffer.map(d => d.value)
    const avg = rates.reduce((a, b) => a + b, 0) / rates.length
    const std = Math.sqrt(
      rates.reduce((sum, r) => sum + Math.pow(r - avg, 2), 0) / rates.length
    )
    const cv = std / avg
    if (avg > 90) return { status: '焦虑', level: 5 }
    if (avg > 80) return { status: '压力大', level: 4 }
    if (avg > 70 && cv < 0.01) return { status: '紧张', level: 3 }
    if (avg < 65 && cv > 0.02) return { status: '深度放松', level: 1 }
    if (avg < 75) return { status: '正常放松', level: 2 }
    return { status: '正常', level: 2 }
  }
}
```

### 11.2 模拟 RRI 生成

```javascript
// soundcare-app/src/pages/player/index.vue:166
const generateSimulatedRRI = (targetHRV, heartRateBPM) => {
  const interval = 60000 / heartRateBPM   // 平均心跳间隔（ms）
  const variation = targetHRV / 2        // 变异范围

  // 每 5 秒生成 5-8 个 RRI
  const count = 5 + Math.floor(Math.random() * 4)
  const rriValues = []
  for (let i = 0; i < count; i++) {
    const noise = (Math.random() - 0.5) * variation
    const rri = interval + noise + (Math.random() - 0.5) * variation * 0.5
    rriValues.push(Math.round(rri))
  }
  return rriValues
}
```
