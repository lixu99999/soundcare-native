## 三、技术方案

### 3.1 核心问题

当前 `hrv-plugin.js`：
```javascript
if (!plugin) {
  return { success: false, errorCode: HRV_ERROR.PLUGIN_NOT_LOADED }
}
```

标准基座下 `uni.requireNativePlugin('SoundCareHRV')` 返回 `null`，所有 HRV 接口**全部失败**，UI 流程跑不通。

### 3.2 解决方案：JS 层 mock 兜底

在 `hrv-plugin.js` 内部检测 `plugin == null` 时，自动切换到 JS 层的 mock 模式：

```javascript
function startMonitoring(options = {}) {
  init()

  if (monitoringActive) {
    return { success: false, errorCode: HRV_ERROR.MONITORING_ALREADY_STARTED }
  }

  if (!plugin) {
    // ✨ 新增：标准基座下启动 JS 层 mock
    startJsMock(options.types)
    monitoringActive = true
    mockModeActive = true
    mockFallbackActive = true  // 标记
    return { success: true, monitoring: true, mockMode: true }
  }

  // ... 原生插件调用（保持不变）
}
```

### 3.3 Mock 数据生成器

```javascript
function startJsMock(types) {
  // 模拟静息态：HR 65 BPM，HRV 50-70ms 之间漂移
  let baseHR = 65
  let baseHRV = 55
  let elapsed = 0

  jsMockTimer = setInterval(() => {
    elapsed += 1
    // 缓慢漂移
    baseHR += (Math.random() - 0.5) * 2
    baseHRV += (Math.random() - 0.5) * 3

    if (types.includes(HRV_TYPE.HEART_RATE)) {
      handleNativeEvent({
        type: HRV_TYPE.HEART_RATE,
        value: Math.round(baseHR),
        timestamp: Date.now(),
        source: 'Mock'  // ← 关键：标记数据来源
      })
    }
    if (types.includes(HRV_TYPE.HRV)) {
      handleNativeEvent({
        type: HRV_TYPE.HRV,
        value: Math.round(baseHRV),
        timestamp: Date.now(),
        source: 'Mock'
      })
    }
  }, 1000)  // 每秒一次
}
```

**关键点**：
- 事件通过同一个 `handleNativeEvent` 路径推送
- 订阅者无感知（不知道是 mock 还是真数据）
- `source: 'Mock'` 字段在 UI 上可见（hrv-monitor 已支持）

## 十一、参考：微信小程序的 HRV Mock 实现

> 同步阅读：`soundcare-app/src/pages/player/index.vue` 第 33-238 行

### 11.1 核心组件

微信小程序用了一个完整的 **`HRVCalculator` 类**（约 120 行），比 v1.1 草案的简单随机游走要精细得多：

```javascript
// soundcare-app/src/pages/player/index.vue:34
class HRVCalculator {
  constructor() {
    this.windowSize = 30 * 1000  // 30秒滚动窗口
    this.rriBuffer = []         // 原始 RRI（毫秒）
    this.heartRateBuffer = []   // 心率历史
  }

  // 从 RRI 计算 RMSSD（标准心率变异算法）
  calculateRMSSD() {
    if (this.rriBuffer.length < 2) return null
    const rris = this.rriBuffer.map(d => d.value)
    let sumSq = 0
    for (let i = 0; i < rris.length - 1; i++) {
      const diff = rris[i + 1] - rris[i]
      sumSq += diff * diff
    }
    return Math.sqrt(sumSq / (rris.length - 1))
  }

  // 从心率变异系数（CV = std/avg）估算状态
  estimateStatusFromHeartRate() {
    if (this.heartRateBuffer.length < 10) return null
    const rates = this.heartRateBuffer.map(d => d.value)
    const avg = rates.reduce((a, b) => a + b, 0) / rates.length
    const std = Math.sqrt(
      rates.reduce((sum, r) => sum + Math.pow(r - avg, 2), 0) / rates.length
    )
    const cv = std / avg
    if (avg > 90) return { status: '焦虑', level: 5 }
    if (avg > 80) return { status: '压力大', level: 4 }
    if (avg > 70 && cv < 0.01) return { status: '紧张', level: 3 }
    if (avg < 65 && cv > 0.02) return { status: '深度放松', level: 1 }
    if (avg < 75) return { status: '正常放松', level: 2 }
    return { status: '正常', level: 2 }
  }
}
```

### 11.2 模拟 RRI 生成

```javascript
// soundcare-app/src/pages/player/index.vue:166
const generateSimulatedRRI = (targetHRV, heartRateBPM) => {
  const interval = 60000 / heartRateBPM   // 平均心跳间隔（ms）
  const variation = targetHRV / 2        // 变异范围

  // 每 5 秒生成 5-8 个 RRI
  const count = 5 + Math.floor(Math.random() * 4)
  const rriValues = []
  for (let i = 0; i < count; i++) {
    const noise = (Math.random() - 0.5) * variation
    const rri = interval + noise + (Math.random() - 0.5) * variation * 0.5
    rriValues.push(Math.round(rri))
  }
  return rriValues
}
```
